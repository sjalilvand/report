     3→// محیط بصری اتصال جداول — بخش ۱۶ سند

     6→// پرس‌وجوی تجمیعی روی مدل اتصالی با گزینه تجمیع پیش از اتصال
     7→
     8→import { useState, useEffect, useCallback, useRef } from 'react'
     9→import { api } from '@/lib/client'
    10→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card'
    11→import { Button } from '@/components/ui/button'
    12→import { Badge } from '@/components/ui/badge'
    13→import { Input } from '@/components/ui/input'
    14→import { Label } from '@/components/ui/label'
    15→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    16→import { Switch } from '@/components/ui/switch'
    17→import { ScrollArea } from '@/components/ui/scroll-area'
    18→import { Sheet, SheetContent, SheetHeader, SheetTitle, SheetDescription } from '@/components/ui/sheet'
    19→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table'
    20→import { Alert, AlertDescription } from '@/components/ui/alert'
    21→import { useToast } from '@/hooks/use-toast'
    22→import { detectGraphIssues, JOIN_TYPE_LABELS, CARDINALITY_LABELS, JoinCheckResult } from '@/lib/join'
    23→import { FieldMeta } from '@/lib/types'
    24→import { formatNumber } from '@/lib/currency'
    25→import {
    26→  Workflow, GripVertical, Link2, Unlink, Sparkles, ShieldAlert, Play, Save,
    27→  Plus, X, Network, Table2, CheckCircle2, TriangleAlert, Layers,
    28→} from 'lucide-react'
    29→
    30→interface DatasetRow { id: string; name: string; sheetName: string; rowCount: number; version: number; status: string; fields: string }
    31→interface ColumnPair { fromFieldId: string; toFieldId: string }
    32→interface RelationDTO {
    33→  id: string; name: string; fromDatasetId: string; toDatasetId: string
    34→  cardinality: string; joinType: string; columnPairs: ColumnPair[]
    35→  status: string; filterDirection: string; lastCheck?: string | null
    36→}
    37→interface SuggestionDTO {
    38→  fromDatasetId: string; toDatasetId: string; fromDatasetName: string; toDatasetName: string
    39→  pairs: ColumnPair[]; score: number; reason: string
    40→}
    41→type JqField = FieldMeta
    42→
    43→const CARD_W = 208
    44→const TYPE_FA: Record<string, string> = {
    45→  text: 'متن', integer: 'عدد', decimal: 'اعشار', amount: 'مبلغ', percent: 'درصد',
    46→  date: 'تاریخ', identifier: 'شناسه', category: 'دسته', boolean: 'منطقی', time: 'زمان', datetime: 'زمان‌تاریخ', geo: 'مکانی',
    47→}
    48→
    49→export default function RelationsView({ projectId, datasets, canEdit }: {
    50→  projectId: string
    51→  datasets: DatasetRow[]
    52→  canEdit: boolean
    53→}) {
    54→  const { toast } = useToast()
    55→  const [rels, setRels] = useState<RelationDTO[]>([])
    56→  const [suggestions, setSuggestions] = useState<SuggestionDTO[]>([])
    57→  const [positions, setPositions] = useState<Record<string, { x: number; y: number }>>({})
    58→  const [dragCard, setDragCard] = useState<string | null>(null)
    59→  const dragOffset = useRef({ dx: 0, dy: 0 })
    60→  const [draft, setDraft] = useState<{ fromDs: string; toDs: string; pairs: ColumnPair[]; joinType: string; name: string } | null>(null)
    61→  const [check, setCheck] = useState<JoinCheckResult | null>(null)
    62→  const [checking, setChecking] = useState(false)
    63→  const [preAgg, setPreAgg] = useState(false)
    64→  const [savingRel, setSavingRel] = useState(false)
    65→  const canvasRef = useRef<HTMLDivElement>(null)
    66→
    67→  // پرس‌وجوی اتصالی
    68→  const [rootId, setRootId] = useState('')
    69→  const [pathRelIds, setPathRelIds] = useState<string[]>([])
    70→  const [jqPreAgg, setJqPreAgg] = useState(true)
    71→  const [jqFields, setJqFields] = useState<JqField[]>([])
    72→  const [jqDim, setJqDim] = useState('')
    73→  const [jqMeasures, setJqMeasures] = useState<{ fieldId: string; agg: string }[]>([])
    74→  const [jqResult, setJqResult] = useState<{ rows: Record<string, unknown>[]; totals: Record<string, unknown>; warnings: string[]; path: string[]; graphIssues: { kind: string; message: string }[] } | null>(null)
    75→  const [jqBusy, setJqBusy] = useState(false)
    76→  const [jqError, setJqError] = useState('')
    77→
    78→  const dsById = new Map(datasets.map(d => [d.id, d]))
    79→  const fieldsOf = (dsId: string): FieldMeta[] => {
    80→    try { return JSON.parse(dsById.get(dsId)?.fields ?? '[]') } catch { return [] }
    81→  }
    82→
    83→  const load = useCallback(async () => {
    84→    try {
    85→      const [r, s] = await Promise.all([
    86→        api.get<RelationDTO[]>(`/api/relations?projectId=${projectId}`),
    87→        api.get<SuggestionDTO[]>(`/api/relations/suggest?projectId=${projectId}`),
    88→      ])
    89→      setRels(r)
    90→      setSuggestions(s)
    91→    } catch { /* بی‌صدا */ }
    92→  }, [projectId])
    93→
    94→  useEffect(() => { load() }, [load])
    95→
    96→  // چیدمان اولیه کارت‌ها
    97→  useEffect(() => {
    98→    setPositions(prev => {
    99→      const next = { ...prev }
   100→      datasets.forEach((d, i) => {
   101→        if (!next[d.id]) {
   102→          const col = i % 2
   103→          const row = Math.floor(i / 2)
   104→          next[d.id] = { x: 16 + col * (CARD_W + 90), y: 16 + row * 240 }
   105→        }
   106→      })
   107→      return next
   108→    })
   109→  }, [datasets])
   110→
   111→  const graphIssues = detectGraphIssues(rels.map(r => ({ id: r.id, fromDatasetId: r.fromDatasetId, toDatasetId: r.toDatasetId, status: r.status })))
   112→
   113→  // ---------- بوم: جابه‌جایی کارت ----------
   114→  const onCanvasDrop = (e: React.DragEvent) => {
   115→    if (!dragCard) return
   116→    const rect = canvasRef.current?.getBoundingClientRect()
   117→    if (!rect) return
   118→    const x = Math.max(0, Math.min(rect.width - CARD_W, e.clientX - rect.left - dragOffset.current.dx))
   119→    const y = Math.max(0, e.clientY - rect.top - dragOffset.current.dy)
   120→    setPositions(p => ({ ...p, [dragCard]: { x, y } }))
   121→    setDragCard(null)
   122→  }
   123→
   124→  // ---------- کشیدن ستون روی ستون ----------
   125→  const onColumnDrop = (targetDs: string, targetFieldId: string, e: React.DragEvent) => {
   126→    e.stopPropagation()
   127→    const raw = e.dataTransfer.getData('application/x-column')
   128→    if (!raw) return
   129→    const src = JSON.parse(raw) as { dsId: string; fieldId: string }
   130→    if (src.dsId === targetDs) {
   131→      toast({ title: 'اتصال جدول به خودش ممکن نیست' })
   132→      return
   133→    }
   134→    const fromDs = src.dsId
   135→    const pair = { fromFieldId: src.fieldId, toFieldId: targetFieldId }
   136→    setDraft(prev => {
   137→      if (prev && prev.fromDs === fromDs && prev.toDs === targetDs) {
   138→        return { ...prev, pairs: [...prev.pairs.filter(p => p.toFieldId !== pair.toFieldId), pair] }
   139→      }
   140→      const fromName = dsById.get(fromDs)?.name ?? ''
   141→      const toName = dsById.get(targetDs)?.name ?? ''
   142→      return { fromDs, toDs: targetDs, pairs: [pair], joinType: 'left', name: `${fromName} ← ${toName}` }
   143→    })
   144→    setCheck(null)
   145→    setPreAgg(false)
   146→  }
   147→
   148→  // ---------- اجرای کنترل‌های پیش از اتصال (۱۶.۳) ----------
   149→  const runCheck = async () => {
   150→    if (!draft) return
   151→    setChecking(true)
   152→    try {
   153→      const res = await api.post<{ check: JoinCheckResult }>('/api/relations/check', {
   154→        fromDatasetId: draft.fromDs,
   155→        toDatasetId: draft.toDs,
   156→        columnPairs: draft.pairs,
   157→        joinType: draft.joinType,
   158→        preAggregate: preAgg,
   159→      })
   160→      setCheck(res.check)
   161→    } catch (e) {
   162→      toast({ title: 'خطا در کنترل اتصال', description: e instanceof Error ? e.message : '', variant: 'destructive' })
   163→    } finally {
   164→      setChecking(false)
   165→    }
   166→  }
   167→
   168→  // ---------- ثبت رابطه ----------
   169→  const saveRelation = async () => {
   170→    if (!draft) return
   171→    setSavingRel(true)
   172→    try {
   173→      const res = await api.post<{ blocked?: boolean; check?: JoinCheckResult; message?: string; graphIssues?: { message: string }[] }>('/api/relations', {
   174→        projectId,
   175→        fromDatasetId: draft.fromDs,
   176→        toDatasetId: draft.toDs,
   177→        columnPairs: draft.pairs,
   178→        joinType: draft.joinType,
   179→        name: draft.name,
   180→        preAggregate: preAgg,
   181→      })
   182→      if (res.blocked) {
   183→        setCheck(res.check ?? null)
   184→        toast({ title: 'اتصال مسدود شد', description: res.message, variant: 'destructive' })
   185→        return
   186→      }
   187→      toast({
   188→        title: 'رابطه ثبت شد',
   189→        description: `درصد تطبیق ${((res.check?.matchPercent ?? 0).toLocaleString('fa-IR'))}٪ — ${res.graphIssues?.length ? 'هشدار گراف روابط را ببینید' : 'بدون مشکل حلقه/ابهام'}`,
   190→      })
   191→      setDraft(null)
   192→      setCheck(null)
   193→      await load()
   194→    } catch (e) {
   195→      toast({ title: 'خطا', description: e instanceof Error ? e.message : '', variant: 'destructive' })
   196→    } finally {
   197→      setSavingRel(false)
   198→    }
   199→  }
   200→
   201→  const deleteRelation = async (id: string) => {
   202→    try {
   203→      await api.del(`/api/relations/${id}`)
   204→      setPathRelIds(p => p.filter(x => x !== id))
   205→      await load()
   206→      toast({ title: 'رابطه حذف شد' })
   207→    } catch (e) {
   208→      toast({ title: 'خطا', description: e instanceof Error ? e.message : '', variant: 'destructive' })
   209→    }
   210→  }
   211→
   212→  const applySuggestion = async (s: SuggestionDTO, accept: boolean) => {
   213→    if (accept) {
   214→      setDraft({ fromDs: s.fromDatasetId, toDs: s.toDatasetId, pairs: s.pairs, joinType: 'left', name: `${s.fromDatasetName} ← ${s.toDatasetName}` })
   215→      setCheck(null)
   216→    } else {
   217→      try {
   218→        await api.post('/api/relations', {
   219→          projectId, fromDatasetId: s.fromDatasetId, toDatasetId: s.toDatasetId,
   220→          columnPairs: s.pairs, joinType: 'left', name: `${s.fromDatasetName} ← ${s.toDatasetName}`,
   221→        })
   222→        toast({ title: 'برای رد پیشنهاد، رابطه ساخته نشد', description: 'پیشنهاد از فهرست پیشنهادها حذف نمی‌شود؛ می‌توانید آن را نادیده بگیرید.' })
   223→      } catch { /* noop */ }
   224→      setSuggestions(prev => prev.filter(x => !(x.fromDatasetId === s.fromDatasetId && x.toDatasetId === s.toDatasetId)))
   225→    }
   226→  }
   227→
   228→  // ---------- پرس‌وجوی اتصالی ----------
   229→  const runJq = async () => {
   230→    if (!rootId) {
   231→      toast({ title: 'جدول پایه را انتخاب کنید', variant: 'destructive' })
   232→      return
   233→    }
   234→    setJqBusy(true)
   235→    setJqError('')
   236→    try {
   237→      const res = await api.post<typeof jqResult & { fields: JqField[]; scannedRecords: number }>('/api/relations/query', {
   238→        rootDatasetId: rootId,
   239→        relationIds: pathRelIds,
   240→        preAggregate: jqPreAgg,
   241→        spec: {
   242→          dimensions: jqDim ? [jqDim] : [],
   243→          measures: jqMeasures.length > 0 ? jqMeasures : [{ agg: 'count' }],
   244→          limit: 50,
   245→        },
   246→      })
   247→      setJqResult(res)
   248→      if (res.fields?.length) setJqFields(res.fields)
   249→    } catch (e) {
   250→      setJqResult(null)
   251→      setJqError(e instanceof Error ? e.message : 'خطا در اجرای پرس‌وجو')
   252→    } finally {
   253→      setJqBusy(false)
   254→    }
   255→  }
   256→
   257→  const activeRels = rels.filter(r => r.status === 'active')
   258→
   259→  // نقاط اتصال خطوط روی بوم
   260→  const anchors = (r: RelationDTO) => {
   261→    const a = positions[r.fromDatasetId]
   262→    const b = positions[r.toDatasetId]
   263→    if (!a || !b) return null
   264→    const ay = a.y + 34
   265→    const by = b.y + 34
   266→    if (a.x + CARD_W <= b.x + 4) return { x1: a.x + CARD_W, y1: ay, x2: b.x, y2: by, label: (a.x + CARD_W + b.x) / 2, labelY: (ay + by) / 2 }
   267→    if (b.x + CARD_W <= a.x + 4) return { x1: a.x, y1: ay, x2: b.x + CARD_W, y2: by, label: (a.x + b.x + CARD_W) / 2, labelY: (ay + by) / 2 }
   268→    return { x1: a.x + CARD_W / 2, y1: a.y, x2: b.x + CARD_W / 2, y2: b.y, label: (a.x + b.x) / 2 + CARD_W / 2, labelY: Math.min(a.y, b.y) - 6 }
   269→  }
   270→
   271→  return (
   272→    <div className="space-y-4">
   273→      <div className="grid gap-4 lg:grid-cols-[1fr_320px]">
   274→        {/* بوم اتصال */}
   275→        <Card>
   276→          <CardHeader className="pb-3">
   277→            <CardTitle className="text-base flex items-center gap-2"><Network className="size-4 text-primary" /> بوم مدل‌سازی رابطه‌ها</CardTitle>
   278→            <CardDescription>
   279→              ستون یک جدول را روی ستون جدول دیگر بکشید تا رابطه ساخته شود؛ کلید مرکب = کشیدن چند ستون هنگام باز بودن پنل رابطه
   280→            </CardDescription>
   281→          </CardHeader>
   282→          <CardContent>
   283→            <div
   284→              ref={canvasRef}
   285→              onDragOver={e => e.preventDefault()}
   286→              onDrop={onCanvasDrop}
   287→              className="relative h-[480px] rounded-lg border border-dashed bg-[radial-gradient(circle,_hsl(var(--border))_1px,_transparent_1px)] [background-size:18px_18px] overflow-hidden"
   288→            >
   289→              <svg className="absolute inset-0 size-full pointer-events-none">
   290→                {activeRels.map(r => {
   291→                  const a = anchors(r)
   292→                  if (!a) return null
   293→                  return (
   294→                    <g key={r.id}>
   295→                      <path
   296→                        d={`M ${a.x1} ${a.y1} C ${(a.x1 + a.x2) / 2} ${a.y1}, ${(a.x1 + a.x2) / 2} ${a.y2}, ${a.x2} ${a.y2}`}
   297→                        fill="none" stroke="hsl(var(--primary))" strokeWidth="2" strokeOpacity="0.65"
   298→                      />
   299→                      <circle cx={a.x1} cy={a.y1} r="4" fill="hsl(var(--primary))" />
   300→                      <circle cx={a.x2} cy={a.y2} r="4" fill="hsl(var(--primary))" />
   301→                      <text x={a.label} y={a.labelY} textAnchor="middle" className="fill-primary text-[10px] font-medium" direction="rtl">
   302→                        {JOIN_TYPE_LABELS[r.joinType as keyof typeof JOIN_TYPE_LABELS]?.split(' ')[0] ?? r.joinType}
   303→                      </text>
   304→                    </g>
   305→                  )
   306→                })}
   307→              </svg>
   308→
   309→              {datasets.map(ds => {
   310→                const pos = positions[ds.id] ?? { x: 0, y: 0 }
   311→                const fields = fieldsOf(ds.id)
   312→                return (
   313→                  <div
   314→                    key={ds.id}
   315→                    className="absolute rounded-xl border bg-card shadow-sm w-52 select-none"
   316→                    style={{ left: pos.x, top: pos.y }}
   317→                  >
   318→                    <div
   319→                      draggable={canEdit}
   320→                      onDragStart={e => {
   321→                        const rect = (e.currentTarget.parentElement as HTMLElement).getBoundingClientRect()
   322→                        dragOffset.current = { dx: e.clientX - rect.left, dy: e.clientY - rect.top }
   323→                        setDragCard(ds.id)
   324→                      }}
   325→                      onDragEnd={() => setDragCard(null)}
   326→                      className="flex items-center gap-1.5 px-2.5 py-2 border-b bg-muted/50 rounded-t-xl cursor-grab active:cursor-grabbing"
   327→                    >
   328→                      <GripVertical className="size-3.5 text-muted-foreground shrink-0" />
   329→                      <div className="min-w-0 flex-1">
   330→                        <p className="text-[11px] font-semibold truncate">{ds.name}</p>
   331→                        <p className="text-[9px] text-muted-foreground">{formatNumber(ds.rowCount)} رکورد</p>
   332→                      </div>
   333→                    </div>
   334→                    <ScrollArea className="h-[136px]">
   335→                      <div className="py-1">
   336→                        {fields.map(f => (
   337→                          <div
   338→                            key={f.id}
   339→                            draggable={canEdit}
   340→                            onDragStart={e => {
   341→                              e.dataTransfer.setData('application/x-column', JSON.stringify({ dsId: ds.id, fieldId: f.id }))
   342→                            }}
   343→                            onDragOver={e => e.preventDefault()}
   344→                            onDrop={e => onColumnDrop(ds.id, f.id, e)}
   345→                            className="flex items-center justify-between gap-1 px-2 py-1 text-[10px] hover:bg-accent/60 cursor-grab active:cursor-grabbing"
   346→                            title="این ستون را بکشید و روی ستون جدول دیگر رها کنید"
   347→                          >
   348→                            <span className="truncate">{f.label}</span>
   349→                            <span className="flex items-center gap-1 shrink-0">
   350→                              {f.role === 'measure' && <span className="text-[8px] rounded bg-amber-100 text-amber-700 px-1">سنجه</span>}
   351→                              {f.role === 'identifier' && <span className="text-[8px] rounded bg-teal-100 text-teal-700 px-1">کلید</span>}
   352→                              <Badge variant="outline" className="text-[8px] px-1 h-4">{TYPE_FA[f.type] ?? f.type}</Badge>
   353→                            </span>
   354→                          </div>
   355→                        ))}
   356→                      </div>
   357→                    </ScrollArea>
   358→                  </div>
   359→                )
   360→              })}
   361→
   362→              {datasets.length === 0 && (
   363→                <div className="absolute inset-0 flex items-center justify-center text-sm text-muted-foreground">
   364→                  ابتدا در برگه «فایل‌ها و مجموعه‌داده» حداقل دو مجموعه‌داده بسازید.
   365→                </div>
   366→              )}
   367→            </div>
   368→          </CardContent>
   369→        </Card>
   370→
   371→        {/* ستون کنار: پیشنهادها، روابط، هشدارهای گراف */}
   372→        <div className="space-y-4">
   373→          {graphIssues.length > 0 && (
   374→            <Alert className="border-amber-300 bg-amber-50">
   375→              <ShieldAlert className="size-4 text-amber-600" />
   376→              <AlertDescription className="text-xs text-amber-800 space-y-1">
   377→                {graphIssues.map((g, i) => <p key={i}>{g.message}</p>)}
   378→              </AlertDescription>
   379→            </Alert>
   380→          )}
   381→
   382→          <Card>
   383→            <CardHeader className="pb-2">
   384→              <CardTitle className="text-sm flex items-center gap-1.5"><Sparkles className="size-3.5 text-primary" /> روابط پیشنهادی سامانه</CardTitle>
   385→            </CardHeader>
   386→            <CardContent className="space-y-2">
   387→              {suggestions.length === 0 && <p className="text-xs text-muted-foreground">پیشنهاد جدیدی نیست؛ بر اساس همخوانی نام ستون‌ها و سازگاری نوع داده پیشنهاد می‌شوند.</p>}
   388→              {suggestions.map((s, i) => (
   389→                <div key={i} className="rounded-lg border p-2.5 space-y-1.5">
   390→                  <p className="text-[11px] font-medium">{s.fromDatasetName} ← {s.toDatasetName}</p>
   391→                  <p className="text-[10px] text-muted-foreground">{s.reason} | اطمینان {Math.round(s.score * 100).toLocaleString('fa-IR')}٪</p>
   392→                  {canEdit && (
   393→                    <div className="flex gap-1.5">
   394→                      <Button size="sm" variant="outline" className="h-6 text-[10px] gap-1" onClick={() => applySuggestion(s, true)}>
   395→                        <CheckCircle2 className="size-3" /> بررسی و پذیرش
   396→                      </Button>
   397→                      <Button size="sm" variant="ghost" className="h-6 text-[10px]" onClick={() => applySuggestion(s, false)}>نادیده بگیر</Button>
   398→                    </div>
   399→                  )}
   400→                </div>
   401→              ))}
   402→            </CardContent>
   403→          </Card>
   404→
   405→          <Card>
   406→            <CardHeader className="pb-2">
   407→              <CardTitle className="text-sm flex items-center gap-1.5"><Link2 className="size-3.5 text-primary" /> روابط ثبت‌شده ({activeRels.length.toLocaleString('fa-IR')})</CardTitle>
   408→            </CardHeader>
   409→            <CardContent className="space-y-2">
